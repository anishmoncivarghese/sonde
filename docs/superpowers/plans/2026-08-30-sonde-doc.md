# `sonde doc` Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Render the code graph as a deterministic, module-level Markdown document a human can read, which reports its own staleness.

**Architecture:** Two new store queries return flat rows; `src/doc/modules.ts` aggregates them into a `ModuleGraph` (pure); `src/doc/render.ts` turns that into Markdown (pure); `src/doc/index.ts` adds the freshness stamp and the write-safety rules. Purity in the middle two is what lets determinism be tested directly rather than through the filesystem.

**Tech Stack:** TypeScript, `better-sqlite3`, `commander`, vitest.

**Spec:** `docs/superpowers/specs/2026-08-30-sonde-doc-design.md`

## Global Constraints

Copied from the spec and `AGENTS.md`. Every task implicitly includes these.

- **Run `nvm use` in every shell before any `node` or `npm` command.**
- **Determinism:** the same repository state MUST produce a byte-identical document. Stable sort every list; no timestamps in the body; no absolute paths; no hash-map iteration order.
- **Stamp the commit SHA, never wall-clock time** (spec §4.1).
- **The stamp must not lie** (spec §4.2): dirty worktree and absent revision each get their own wording.
- **Never fabricate** (invariant 1). Dependencies carry tier counts; heuristic ones are visibly distinguished.
- **Degrade with a warning, never fail silently** (invariant 8).
- **All repository reads and writes go through `src/repo/boundary.ts`** (invariant 6).
- **Never clobber a file Sonde did not write** (spec §5.1), following `sonde init`'s precedent.
- **TDD**, conventional commits, one commit per task.

Add to the `Store` class in `src/store/repos.ts`:

```ts
  /**
   * Flat edge rows for documentation aggregation.
   *
   * Deliberately ungrouped: grouping happens in pure code (src/doc/modules.ts)
   * so that the determinism the document depends on can be tested without a
   * database (spec §4.1).
   *
   * CONTAINS is parent-to-child inside one file and carries no dependency
   * signal. TESTS is always HEURISTIC and would require the invariant 7
   * disclosure this version does not render.
   */
  docEdgeRows(): Array<{
    srcFile: string;
    dstFile: string;
    dstName: string;
    kind: EdgeKind;
    tier: string;
  }> {
    return this.db
      .prepare(
        `SELECT src_file.path      AS srcFile,
                dst_file.path      AS dstFile,
                target.short_name  AS dstName,
                edge.kind          AS kind,
                edge.tier          AS tier
         FROM edge
         JOIN symbol AS source   ON source.id = edge.src_symbol_id
         JOIN symbol AS target   ON target.id = edge.dst_symbol_id
         JOIN file   AS src_file ON src_file.id = source.file_id
         JOIN file   AS dst_file ON dst_file.id = target.file_id
         WHERE edge.kind NOT IN ('CONTAINS', 'TESTS')`,
      )
      .all() as Array<{
        srcFile: string; dstFile: string; dstName: string;
        kind: EdgeKind; tier: string;
      }>;
  }

  docSymbolCounts(): Array<{ filePath: string; symbols: number }> {
    return this.db
      .prepare(
        `SELECT file.path AS filePath, COUNT(symbol.id) AS symbols
         FROM file
         LEFT JOIN symbol ON symbol.file_id = file.id
         GROUP BY file.path`,
      )
      .all() as Array<{ filePath: string; symbols: number }>;
  }

  /**
   * The store exposes only `hasParseFailures(): boolean`. The document reports
   * how many files failed, and printing a boolean as a number would fabricate
   * a statistic in an artifact whose whole premise is honesty.
   */
  countParseFailures(): number {
    const row = this.db
      .prepare("SELECT COUNT(*) AS n FROM file WHERE parse_state != 'ok'")
      .get() as { n: number };
    return row.n;
  }
```

Column names verified against `src/store/schema.sql`: `symbol` has **`file_id`**, not a file path, hence the two extra joins to `file`. `file.parse_state` defaults to `'ok'`.

// src/doc/modules.ts

export type DocTier = "COMPILER" | "LEXICAL" | "HEURISTIC";

export interface ModuleDependency {
  from: string;
  to: string;
  total: number;
  byTier: Record<DocTier, number>;
}

export interface ModuleSummary {
  path: string;
  files: number;
  symbols: number;
  surface: string[];
}

export interface ModuleGraph {
  modules: ModuleSummary[];
  dependencies: ModuleDependency[];
  tierTotals: Record<DocTier, number>;
}

interface EdgeInput {
  srcFile: string; dstFile: string; dstName: string; kind: string; tier: string;
}

const TIERS: DocTier[] = ["COMPILER", "LEXICAL", "HEURISTIC"];

function zeroTiers(): Record<DocTier, number> {
  return { COMPILER: 0, LEXICAL: 0, HEURISTIC: 0 };
}

function isDocTier(tier: string): tier is DocTier {
  return (TIERS as string[]).includes(tier);
}

/**
 * spec §3.1: a module is a source file's immediate parent directory.
 *
 * Mechanical and language-agnostic. A root-level file is named "." rather than
 * the empty string, so it can never render as a blank row.
 */
export function moduleOf(filePath: string): string {
  const cut = filePath.lastIndexOf("/");
  return cut <= 0 ? "." : filePath.slice(0, cut);
}

export function buildModuleGraph(
  edges: EdgeInput[],
  symbolCounts: Array<{ filePath: string; symbols: number }>,
): ModuleGraph {
  const files = new Map<string, Set<string>>();
  const symbols = new Map<string, number>();
  for (const row of symbolCounts) {
    const module = moduleOf(row.filePath);
    (files.get(module) ?? files.set(module, new Set()).get(module)!).add(row.filePath);
    symbols.set(module, (symbols.get(module) ?? 0) + row.symbols);
  }

  const deps = new Map<string, ModuleDependency>();
  const surface = new Map<string, Set<string>>();
  const tierTotals = zeroTiers();

  for (const row of edges) {
    const from = moduleOf(row.srcFile);
    const to = moduleOf(row.dstFile);
    // spec §3.2: intra-module coupling is not architecture.
    if (from === to) continue;
    if (!isDocTier(row.tier)) continue;

    tierTotals[row.tier] += 1;

    const key = `${from} ${to}`;
    const dep = deps.get(key) ?? { from, to, total: 0, byTier: zeroTiers() };
    dep.total += 1;
    dep.byTier[row.tier] += 1;
    deps.set(key, dep);

    // spec §3.3: surface is what other modules actually reference.
    (surface.get(to) ?? surface.set(to, new Set()).get(to)!).add(row.dstName);
  }

  const modulePaths = new Set<string>([
    ...files.keys(), ...surface.keys(),
    ...[...deps.values()].flatMap((d) => [d.from, d.to]),
  ]);

  // Every list is sorted: the document must be byte-identical across runs
  // regardless of database or map iteration order (spec §4.1).
  return {
    modules: [...modulePaths].sort().map((path) => ({
      path,
      files: files.get(path)?.size ?? 0,
      symbols: symbols.get(path) ?? 0,
      surface: [...(surface.get(path) ?? [])].sort(),
    })),
    dependencies: [...deps.values()].sort(
      (a, b) => a.from.localeCompare(b.from) || a.to.localeCompare(b.to),
    ),
    tierTotals,
  };
}
```

```ts
// src/doc/render.ts
import type { ModuleDependency, ModuleGraph } from "./modules.js";

/** Identifies a file as Sonde's, so the writer never clobbers a human's (spec §5.1). */
export const DOC_MARKER =
  "<!-- generated by sonde doc; edits will be overwritten -->";

export interface DocStamp {
  revision: string | null;
  dirty: boolean | null;
  driftedFiles: number;
  parseFailures: number;
}

/**
 * spec §4.2: the stamp must not lie.
 *
 * A dirty worktree stamped with a bare SHA would claim to describe a commit it
 * does not describe, and `gitState` returns null when git could not be
 * consulted at all -- which must never read as clean (invariant 8).
 */
function stampLine(stamp: DocStamp): string {
  if (stamp.revision === null) return "Describes an unversioned working tree.";
  const suffix = stamp.dirty === true
    ? " plus uncommitted changes"
    : stamp.dirty === null
      ? " (git state unknown)"
      : "";
  return `Describes commit ${stamp.revision}${suffix}.`;
}

/** A dependency with no resolved evidence is inference, and renders dashed. */
function isInferred(dep: ModuleDependency): boolean {
  return dep.byTier.COMPILER === 0 && dep.byTier.LEXICAL === 0;
}

function mermaidId(path: string): string {
  return path.replace(/[^A-Za-z0-9]/g, "_");
}

function evidence(dep: ModuleDependency): string {
  return (["COMPILER", "LEXICAL", "HEURISTIC"] as const)
    .filter((tier) => dep.byTier[tier] > 0)
    .map((tier) => `${tier} ${dep.byTier[tier]}`)
    .join(", ");
}

export function renderDoc(graph: ModuleGraph, stamp: DocStamp): string {
  const lines: string[] = [];
  const total = graph.tierTotals.COMPILER + graph.tierTotals.LEXICAL +
    graph.tierTotals.HEURISTIC;

  lines.push("# Architecture", "", DOC_MARKER, "");
  lines.push(stampLine(stamp), "");

  if (stamp.driftedFiles > 0) {
    lines.push(
      `> **${stamp.driftedFiles} file(s) have changed since this was generated.** ` +
      "Run `sonde doc` to refresh.",
      "",
    );
  }
  if (stamp.parseFailures > 0) {
    lines.push(
      `> **${stamp.parseFailures} file(s) did not parse cleanly**, so their ` +
      "symbols may be missing here.",
      "",
    );
  }

  lines.push(
    "Cross-module references by how they were resolved: " +
    `COMPILER ${graph.tierTotals.COMPILER}, LEXICAL ${graph.tierTotals.LEXICAL}, ` +
    `HEURISTIC ${graph.tierTotals.HEURISTIC}.`,
    "",
    "`HEURISTIC` means the target was inferred from a name rather than resolved. " +
    "Dependencies supported only by heuristic references are drawn with dashed " +
    "arrows below and should be read as likely, not certain." +
    (graph.tierTotals.COMPILER === 0
      ? " Run `sonde index --resolve` for compiler-resolved edges."
      : ""),
    "",
  );

  lines.push("## Module dependencies", "");
  if (graph.dependencies.length === 0) {
    lines.push("No cross-module references were found.", "");
  } else {
    lines.push("```mermaid", "graph LR");
    for (const dep of graph.dependencies) {
      const arrow = isInferred(dep) ? "-.->" : "-->";
      lines.push(
        `  ${mermaidId(dep.from)}["${dep.from}"] ${arrow}|${dep.total}| ` +
        `${mermaidId(dep.to)}["${dep.to}"]`,
      );
    }
    lines.push("```", "");
    lines.push("| From | To | References | Evidence |", "|---|---|---:|---|");
    for (const dep of graph.dependencies) {
      lines.push(`| \`${dep.from}\` | \`${dep.to}\` | ${dep.total} | ${evidence(dep)} |`);
    }
    lines.push("");
  }

  lines.push("## Modules", "");
  lines.push("| Module | Files | Symbols | Used by other modules |", "|---|---:|---:|---|");
  for (const module of graph.modules) {
    const surface = module.surface.length === 0
      ? "—"
      : module.surface.map((name) => `\`${name}\``).join(", ");
    lines.push(
      `| \`${module.path}\` | ${module.files} | ${module.symbols} | ${surface} |`,
    );
  }
  lines.push("");
  lines.push(
    `Symbol-level detail is not committed, because it changes on every edit. ` +
    "Run `sonde doc --module <path>` for it.",
    "",
  );
  if (total === 0) lines.push("No references were indexed.", "");

  return lines.join("\n");
}

// src/doc/index.ts
import type { RepoBoundary } from "../repo/boundary.js";
import type { Store } from "../store/index.js";
import { gitState } from "../repo/git.js";
import { checkDrift } from "../index/drift.js";
import { buildModuleGraph } from "./modules.js";
import { DOC_MARKER, renderDoc, type DocStamp } from "./render.js";

export const DOC_PATH = "ARCHITECTURE.md";

export type WriteOutcome =
  | { action: "created" }
  | { action: "updated" }
  | { action: "unchanged" }
  | { action: "refused"; reason: "not-generated-by-sonde" };

export function generateDoc(boundary: RepoBoundary, store: Store): string {
  const graph = buildModuleGraph(store.docEdgeRows(), store.docSymbolCounts());
  const git = gitState(boundary);
  const drift = checkDrift(boundary, store);
  const stamp: DocStamp = {
    revision: git.revision,
    dirty: git.dirty,
    driftedFiles: drift.driftCount,
    parseFailures: store.countParseFailures(),
  };
  return renderDoc(graph, stamp);
}

/** spec §5.1: never clobber a file Sonde did not write. */
export function writeDoc(boundary: RepoBoundary, content: string): WriteOutcome {
  let existing: string | null = null;
  try {
    existing = boundary.readFile(DOC_PATH).toString("utf8");
  } catch (error) {
    // Only a missing file means "create". A permission error must not be
    // mistaken for absence and silently overwrite (invariant 8).
    if ((error as NodeJS.ErrnoException).code !== "ENOENT") throw error;
  }

  if (existing === null) {
    boundary.writeFile(DOC_PATH, content);
    return { action: "created" };
  }
  if (!existing.includes(DOC_MARKER)) {
    return { action: "refused", reason: "not-generated-by-sonde" };
  }
  if (existing === content) return { action: "unchanged" };
  boundary.writeFile(DOC_PATH, content);
  return { action: "updated" };
}
